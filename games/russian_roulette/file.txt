  <script>
    /* ==========================================================
       1. SCENE, CAMERA, & LIGHTING
       ========================================================== */
    const container = document.getElementById('canvas-container');
    const scene = new THREE.Scene();
    scene.fog = new THREE.FogExp2(0x0e1117, 0.04);

    const camera = new THREE.PerspectiveCamera(40, window.innerWidth / window.innerHeight, 0.1, 100);
    // Adjusted camera angle for an unobstructed 3/4 view of the cylinder
    camera.position.set(3.4, 1.4, 3.8);

    const renderer = new THREE.WebGLRenderer({ antialias: true });
    renderer.setSize(window.innerWidth, window.innerHeight);
    renderer.setPixelRatio(Math.min(window.devicePixelRatio, 2));
    renderer.shadowMap.enabled = true;
    renderer.shadowMap.type = THREE.PCFSoftShadowMap;
    renderer.toneMapping = THREE.ACESFilmicToneMapping;
    renderer.toneMappingExposure = 1.15;
    container.appendChild(renderer.domElement);

    const controls = new THREE.OrbitControls(camera, renderer.domElement);
    controls.enableDamping = true;
    controls.dampingFactor = 0.05;
    controls.target.set(0, 0.05, -0.1);

    // Multi-angle lighting to accentuate cylinder curves & flutes
    const ambientLight = new THREE.AmbientLight(0xffffff, 0.85);
    scene.add(ambientLight);

    const keyLight = new THREE.DirectionalLight(0xffffff, 2.8);
    keyLight.position.set(5, 7, 5);
    keyLight.castShadow = true;
    scene.add(keyLight);

    const fillLight = new THREE.DirectionalLight(0x90b4ce, 1.5);
    fillLight.position.set(-6, 3, 2);
    scene.add(fillLight);

    const rimLight = new THREE.DirectionalLight(0x38bdf8, 2.0);
    rimLight.position.set(0, 5, -5);
    scene.add(rimLight);

    const flashLight = new THREE.PointLight(0xff7700, 0, 15);
    scene.add(flashLight);

    /* ==========================================================
       2. MATERIALS (High-contrast metallic & wood)
       ========================================================== */
    // Dark Blued Frame
    const frameMaterial = new THREE.MeshStandardMaterial({
      color: 0x1f232a,
      metalness: 0.9,
      roughness: 0.35,
    });

    // Bright Polished Stainless Steel (used on cylinder for maximum visibility)
    const cylinderMaterial = new THREE.MeshStandardMaterial({
      color: 0x6e7683,
      metalness: 0.95,
      roughness: 0.22,
    });

    const fluteDarkMaterial = new THREE.MeshStandardMaterial({
      color: 0x181a1f,
      metalness: 0.8,
      roughness: 0.5,
    });

    const brassCartridge = new THREE.MeshStandardMaterial({
      color: 0xcca034,
      metalness: 0.98,
      roughness: 0.2,
    });

    const woodGripMaterial = new THREE.MeshStandardMaterial({
      color: 0x421b0e,
      roughness: 0.65,
      metalness: 0.05,
    });

    /* ==========================================================
       3. REVOLVER MODEL WITH OPEN CYLINDER WINDOW
       ========================================================== */
    const gunRoot = new THREE.Group();
    scene.add(gunRoot);

    // --- REAR FRAME & RECOIL SHIELD (Behind the Cylinder) ---
    const recoilShield = new THREE.Mesh(new THREE.BoxGeometry(0.5, 1.05, 0.35), frameMaterial);
    recoilShield.position.set(0, 0.05, 0.72);
    gunRoot.add(recoilShield);

    // Curved recoil shield cheeks on both sides
    const shieldCheekGeo = new THREE.CylinderGeometry(0.48, 0.48, 0.2, 16);
    const leftShield = new THREE.Mesh(shieldCheekGeo, frameMaterial);
    leftShield.rotation.z = Math.PI / 2;
    leftShield.position.set(0.18, 0.1, 0.62);
    gunRoot.add(leftShield);

    // --- TOP STRAP (Runs above the Cylinder) ---
    const topStrap = new THREE.Mesh(new THREE.BoxGeometry(0.44, 0.16, 1.6), frameMaterial);
    topStrap.position.set(0, 0.63, -0.05);
    gunRoot.add(topStrap);

    // Rear sight notch on top strap
    const rearSight = new THREE.Mesh(new THREE.BoxGeometry(0.12, 0.08, 0.25), frameMaterial);
    rearSight.position.set(0, 0.73, 0.6);
    gunRoot.add(rearSight);

    // --- BOTTOM FRAME / CRANE HOUSING (Runs below the Cylinder) ---
    const bottomFrame = new THREE.Mesh(new THREE.BoxGeometry(0.44, 0.18, 1.45), frameMaterial);
    bottomFrame.position.set(0, -0.52, -0.05);
    gunRoot.add(bottomFrame);

    // --- FRONT FRAME JUNCTION (In front of the Cylinder) ---
    const frontFrame = new THREE.Mesh(new THREE.BoxGeometry(0.46, 1.0, 0.28), frameMaterial);
    frontFrame.position.set(0, 0.05, -0.85);
    gunRoot.add(frontFrame);

    // Notice: The space between Z: -0.7 and Z: 0.55 is completely hollowed out!

    // ==========================================================
    // --- 4. EXPOSED CYLINDER ASSEMBLY ---
    // ==========================================================
    const cylinderGroup = new THREE.Group();
    // Centered directly inside the open window
    cylinderGroup.position.set(0, 0.05, -0.07);
    gunRoot.add(cylinderGroup);

    // Main Cylinder Drum
    const cylBody = new THREE.Mesh(
      new THREE.CylinderGeometry(0.62, 0.62, 1.25, 36),
      cylinderMaterial
    );
    cylBody.rotation.x = Math.PI / 2;
    cylinderGroup.add(cylBody);

    // Central Extractor Pin
    const axisPin = new THREE.Mesh(
      new THREE.CylinderGeometry(0.12, 0.12, 1.4, 16),
      frameMaterial
    );
    axisPin.rotation.x = Math.PI / 2;
    cylinderGroup.add(axisPin);

    // 6 Visible Flutes & Brass Cartridge Heads
    for (let i = 0; i < 6; i++) {
      const angle = (i * Math.PI) / 3;
      const r = 0.36;

      // Brass bullet rims visible at the back of the cylinder
      const cartridge = new THREE.Mesh(
        new THREE.CylinderGeometry(0.13, 0.13, 0.06, 16),
        brassCartridge
      );
      cartridge.rotation.x = Math.PI / 2;
      cartridge.position.set(Math.cos(angle) * r, Math.sin(angle) * r, 0.63);
      cylinderGroup.add(cartridge);

      // Primer center cap
      const primer = new THREE.Mesh(
        new THREE.CylinderGeometry(0.05, 0.05, 0.08, 12),
        cylinderMaterial
      );
      primer.rotation.x = Math.PI / 2;
      primer.position.set(Math.cos(angle) * r, Math.sin(angle) * r, 0.635);
      cylinderGroup.add(primer);

      // Distinct flutes (grooves) cut along the outside circumference
      const fluteAngle = angle + Math.PI / 6;
      const flute = new THREE.Mesh(
        new THREE.CylinderGeometry(0.13, 0.13, 0.85, 12),
        fluteDarkMaterial
      );
      flute.rotation.x = Math.PI / 2;
      flute.position.set(Math.cos(fluteAngle) * 0.58, Math.sin(fluteAngle) * 0.58, -0.05);
      cylinderGroup.add(flute);
    }

    // --- BARREL & SIGHTS ---
    const barrelLength = 3.2;
    const barrel = new THREE.Mesh(
      new THREE.CylinderGeometry(0.24, 0.25, barrelLength, 24),
      frameMaterial
    );
    barrel.rotation.x = Math.PI / 2;
    barrel.position.set(0, 0.35, -(barrelLength / 2) - 0.99);
    gunRoot.add(barrel);

    // Full Underlug / Ejector Shroud
    const underlug = new THREE.Mesh(new THREE.BoxGeometry(0.22, 0.34, barrelLength), frameMaterial);
    underlug.position.set(0, 0.1, -(barrelLength / 2) - 0.99);
    gunRoot.add(underlug);

    // Ejector Rod Tip
    const ejectorRod = new THREE.Mesh(
      new THREE.CylinderGeometry(0.06, 0.06, 0.7, 16),
      cylinderMaterial
    );
    ejectorRod.rotation.x = Math.PI / 2;
    ejectorRod.position.set(0, 0.05, -1.35);
    gunRoot.add(ejectorRod);

    // Front Sight Blade
    const frontSight = new THREE.Mesh(new THREE.BoxGeometry(0.08, 0.22, 0.35), frameMaterial);
    frontSight.position.set(0, 0.68, -barrelLength - 0.75);
    gunRoot.add(frontSight);

    // Red Front Sight Insert
    const redInsert = new THREE.Mesh(
      new THREE.BoxGeometry(0.082, 0.08, 0.12),
      new THREE.MeshBasicMaterial({ color: 0xef4444 })
    );
    redInsert.position.set(0, 0.72, -barrelLength - 0.82);
    gunRoot.add(redInsert);

    // Hollow Muzzle Bore
    const bore = new THREE.Mesh(
      new THREE.CylinderGeometry(0.14, 0.14, 0.2, 16),
      new THREE.MeshBasicMaterial({ color: 0x050505 })
    );
    bore.rotation.x = Math.PI / 2;
    bore.position.set(0, 0.35, -barrelLength - 0.95);
    gunRoot.add(bore);

    // --- TRIGGER GUARD & MOVING TRIGGER ---
    const guardCurve = new THREE.TorusGeometry(0.48, 0.05, 12, 24, Math.PI * 0.85);
    const guard = new THREE.Mesh(guardCurve, frameMaterial);
    guard.position.set(0, -0.58, 0.12);
    guard.rotation.y = Math.PI / 2;
    guard.rotation.x = 0.35;
    gunRoot.add(guard);

    // Trigger Pivot
    const triggerPivot = new THREE.Group();
    triggerPivot.position.set(0, -0.38, 0.22);
    gunRoot.add(triggerPivot);

    const triggerBlade = new THREE.Mesh(new THREE.BoxGeometry(0.08, 0.44, 0.15), cylinderMaterial);
    triggerBlade.position.set(0, -0.15, -0.05);
    triggerBlade.rotation.x = 0.4;
    triggerPivot.add(triggerBlade);

    // --- MOVING HAMMER ---
    const hammerPivot = new THREE.Group();
    hammerPivot.position.set(0, 0.42, 0.82);
    gunRoot.add(hammerPivot);

    const hammerBody = new THREE.Mesh(new THREE.BoxGeometry(0.14, 0.46, 0.24), cylinderMaterial);
    hammerBody.position.set(0, 0.16, 0.06);
    hammerBody.rotation.x = -0.45;
    hammerPivot.add(hammerBody);

    // Serrated Hammer Spur (Thumb cocking pad)
    const hammerSpur = new THREE.Mesh(new THREE.BoxGeometry(0.16, 0.1, 0.18), cylinderMaterial);
    hammerSpur.position.set(0, 0.32, 0.15);
    hammerPivot.add(hammerSpur);

    // --- WOODEN GRIP ---
    const gripGroup = new THREE.Group();
    gripGroup.position.set(0, -0.42, 0.95);
    gripGroup.rotation.x = -0.48;

    const gripMain = new THREE.Mesh(new THREE.BoxGeometry(0.56, 1.85, 0.78), woodGripMaterial);
    gripMain.position.set(0, -0.7, 0);
    gripGroup.add(gripMain);

    // Grip Medallion
    const medallion = new THREE.Mesh(
      new THREE.CylinderGeometry(0.08, 0.08, 0.58, 16),
      brassCartridge
    );
    medallion.rotation.z = Math.PI / 2;
    medallion.position.set(0, -0.45, 0);
    gripGroup.add(medallion);

    // Grip Butt Cap
    const gripButt = new THREE.Mesh(new THREE.SphereGeometry(0.38, 16, 16), frameMaterial);
    gripButt.scale.set(0.72, 0.3, 1.05);
    gripButt.position.set(0, -1.65, 0.05);
    gripGroup.add(gripButt);

    gunRoot.add(gripGroup);

    // --- MUZZLE FLASH ---
    const flashMat = new THREE.MeshBasicMaterial({ color: 0xffd166, transparent: true, opacity: 0 });
    const flashMesh = new THREE.Mesh(new THREE.ConeGeometry(0.48, 1.8, 8), flashMat);
    flashMesh.rotation.x = -Math.PI / 2;
    flashMesh.position.set(0, 0.35, -5.1);
    gunRoot.add(flashMesh);
    flashLight.position.set(0, 0.35, -4.8);

    /* ==========================================================
       5. SYNTHESIZED WEB AUDIO (Mechanical Clicks & Bang)
       ========================================================== */
    const AudioCtx = window.AudioContext || window.webkitAudioContext;
    let audioCtx = null;

    function initAudio() {
      if (!audioCtx) audioCtx = new AudioCtx();
      if (audioCtx.state === 'suspended') audioCtx.resume();
    }

    function playCylinderClick() {
      if (!audioCtx) return;
      const osc = audioCtx.createOscillator();
      const gain = audioCtx.createGain();
      osc.type = 'triangle';
      osc.frequency.setValueAtTime(950, audioCtx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(120, audioCtx.currentTime + 0.04);
      gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.04);
      osc.connect(gain);
      gain.connect(audioCtx.destination);
      osc.start();
      osc.stop(audioCtx.currentTime + 0.04);
    }

    function playShotSound() {
      if (!audioCtx) return;

      // Noise generator for gun blast
      const bufferSize = audioCtx.sampleRate * 0.45;
      const buffer = audioCtx.createBuffer(1, bufferSize, audioCtx.sampleRate);
      const output = buffer.getChannelData(0);
      for (let i = 0; i < bufferSize; i++) {
        output[i] = Math.random() * 2 - 1;
      }

      const noise = audioCtx.createBufferSource();
      noise.buffer = buffer;

      const filter = audioCtx.createBiquadFilter();
      filter.type = 'lowpass';
      filter.frequency.setValueAtTime(2600, audioCtx.currentTime);
      filter.frequency.exponentialRampToValueAtTime(70, audioCtx.currentTime + 0.4);

      const gain = audioCtx.createGain();
      gain.gain.setValueAtTime(1.0, audioCtx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.45);

      noise.connect(filter);
      filter.connect(gain);
      gain.connect(audioCtx.destination);

      // Low frequency punch
      const kick = audioCtx.createOscillator();
      const kickGain = audioCtx.createGain();
      kick.frequency.setValueAtTime(150, audioCtx.currentTime);
      kick.frequency.exponentialRampToValueAtTime(28, audioCtx.currentTime + 0.22);
      kickGain.gain.setValueAtTime(0.9, audioCtx.currentTime);
      kickGain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.22);

      kick.connect(kickGain);
      kickGain.connect(audioCtx.destination);

      noise.start();
      kick.start();
      noise.stop(audioCtx.currentTime + 0.45);
      kick.stop(audioCtx.currentTime + 0.22);
    }

    /* ==========================================================
       6. TRIGGER & FIRING SEQUENCE
       ========================================================== */
    const state = {
      isFiring: false,
      recoilRot: 0,
      recoilZ: 0,
    };

    function triggerPull() {
      if (state.isFiring) return;
      initAudio();
      state.isFiring = true;

      const startTime = performance.now();
      const pullDuration = 240; // Milliseconds for cocking & cylinder index
      const startCylRot = cylinderGroup.rotation.z;
      const targetCylRot = startCylRot + Math.PI / 3; // 60 degree rotation per chamber

      playCylinderClick();

      function animatePull(time) {
        const elapsed = time - startTime;
        const p = Math.min(elapsed / pullDuration, 1);
        const smoothP = p * p * (3 - 2 * p);

        // 1. Pull trigger back
        triggerPivot.rotation.x = -smoothP * 0.45;

        // 2. Cock hammer rearward
        hammerPivot.rotation.x = -smoothP * 0.85;

        // 3. Actively rotate the exposed cylinder
        cylinderGroup.rotation.z = startCylRot + (targetCylRot - startCylRot) * smoothP;

        if (p < 1) {
          requestAnimationFrame(animatePull);
        } else {
          fire();
        }
      }

      requestAnimationFrame(animatePull);
    }

    function fire() {
      playShotSound();

      // Hammer strikes firing pin & trigger resets
      hammerPivot.rotation.x = 0.12;
      triggerPivot.rotation.x = 0.0;

      // Recoil impulses
      state.recoilRot = 0.42;
      state.recoilZ = 0.28;

      // Flash illumination
      flashMat.opacity = 1.0;
      flashLight.intensity = 8.0;
      flashMesh.scale.set(1 + Math.random() * 0.5, 1 + Math.random() * 0.5, 1 + Math.random() * 0.5);

      setTimeout(() => {
        flashMat.opacity = 0;
        flashLight.intensity = 0;
      }, 50);

      // Return hammer to resting battery position
      setTimeout(() => {
        hammerPivot.rotation.x = 0;
        state.isFiring = false;
      }, 190);
    }

    /* ==========================================================
       7. INTERACTION & ANIMATION LOOP
       ========================================================== */
    window.addEventListener('pointerdown', (e) => {
      if (e.button === 0) triggerPull();
    });

    window.addEventListener('resize', () => {
      camera.aspect = window.innerWidth / window.innerHeight;
      camera.updateProjectionMatrix();
      renderer.setSize(window.innerWidth, window.innerHeight);
    });

    let lastTime = performance.now();

    function animate() {
      requestAnimationFrame(animate);

      const now = performance.now();
      const delta = Math.min((now - lastTime) / 1000, 0.1);
      lastTime = now;

      // Spring-damped recoil recovery
      state.recoilRot += (0 - state.recoilRot) * 15 * delta;
      state.recoilZ += (0 - state.recoilZ) * 15 * delta;

      gunRoot.rotation.x = state.recoilRot;
      gunRoot.position.z = state.recoilZ;

      // Subtle breathing motion when idle
      if (!state.isFiring) {
        gunRoot.position.y = Math.sin(now * 0.002) * 0.02;
      }

      controls.update();
      renderer.render(scene, camera);
    }

    animate();
  </script>
